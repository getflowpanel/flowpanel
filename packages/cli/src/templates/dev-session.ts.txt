// DEVELOPMENT STUB — REPLACE BEFORE PRODUCTION. Every caller is an admin; swap in your auth provider or devAuth() from "@flowpanel/kit".

// Read off globalThis so the stub compiles even where @types/node is absent.
const nodeEnv = (globalThis as { process?: { env?: { NODE_ENV?: string } } }).process?.env?.NODE_ENV;

export async function getSession(): Promise<{ id: string; user: { id: string; role: string } }> {
  if (nodeEnv === "production") {
    throw new Error(
      "flowpanel: the development session stub cannot run in production — it signs " +
        "every request in as an admin. Replace getSession() in this file with your " +
        "real auth provider.",
    );
  }
  console.warn(
    "⚠ flowpanel: development session stub in use — every request is treated as an admin.\n" +
      "  Replace getSession() in this file before deploying.",
  );
  return { id: "dev", user: { id: "dev", role: "admin" } };
}
