import type { Config } from "tailwindcss";

/**
 * FlowPanel design tokens are CSS custom properties defined in
 * `styles/admin.css` (`--fp-bg-1`, `--fp-text-1`, …). On Tailwind v4 the
 * `@theme {}` block inside `admin.css` registers the utilities directly.
 * On Tailwind v3 we mirror the token map into `theme.extend` here so
 * `bg-fp-bg-1`, `text-fp-text-1`, `border-fp-border-1`, `rounded-fp`,
 * `shadow-fp-sm`, etc. actually emit CSS.
 */
const fpColor = (name: string) => `hsl(var(--fp-${name}) / <alpha-value>)`;

  content: [
    "./app/**/*.{ts,tsx}",
    "./src/**/*.{ts,tsx}",
    // Only `@flowpanel/kit` is a direct dependency; the packages carrying the
    // class strings sit in pnpm's virtual store, hence the second glob.

    "./node_modules/.pnpm/node_modules/@flowpanel/*/dist/**/*.{js,mjs}",
  ],
  theme: {
