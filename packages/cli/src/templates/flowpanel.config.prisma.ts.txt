import { prisma } from "{{DB}}";

// Augment "@flowpanel/kit": it is the package you installed, and the
// interfaces below merge into the same declarations "@flowpanel/core" owns.
declare module "@flowpanel/kit" {
  interface FlowpanelTypes {
    db: typeof prisma;
  }
  // resource name -> row type. A Prisma resource is a bare model-name string,
  // so this map is the only thing that gives it a row type: without an entry
  // here, columns and filters fall back to unchecked strings.
  interface FlowpanelResources {
    // User: import("@prisma/client").User;
  }
}

export default defineAdmin({
  adapter: prismaAdapter({ prisma }),

  auth: {
    session: getSession,
    role: (s) =>

    requireRole: "admin",
  },

  realtime: { driver: "memory" },

  theme: {
    brand: { name: "{{APP_NAME}}" },
    accent: "220 90% 50%",
    mode: "auto",
  },

  resources: [
