/* FlowPanel design tokens — 30 CSS custom properties + motion + theme map */

/* @source paths are relative to this file, not the app root — this file can
   land at `styles/admin.css` or `src/styles/admin.css` depending on scaffold
   layout, so `init` fills in {{SOURCE_UP}} with the right number of `../`
   segments back to the app root where node_modules lives. Without these,
   Tailwind v4 never scans the @flowpanel packages, so utility classes used
   inside their components (e.g. `bg-fp-bg-1`) are never emitted. */
@source "{{SOURCE_UP}}node_modules/@flowpanel/react/dist";
@source "{{SOURCE_UP}}node_modules/@flowpanel/next/dist";
@source "{{SOURCE_UP}}node_modules/@flowpanel/charts/dist";
@source "{{SOURCE_UP}}node_modules/flowpanel/dist";

:root {
  /* Color (HSL triplets) */
  --fp-bg-1: 0 0% 100%;
  --fp-bg-2: 220 14% 96%;
  --fp-bg-3: 220 14% 92%;
  --fp-text-1: 220 10% 15%;
  --fp-text-2: 220 10% 40%;
  /* 45% (not 60%) so muted text on --fp-bg-1/2 clears WCAG AA 4.5:1. */
  --fp-text-3: 220 10% 45%;
  --fp-border-1: 220 13% 91%;
  --fp-border-2: 220 13% 86%;
  /* 50% (not 56%) so accent-text (#f5f8ff) on accent clears WCAG AA 4.5:1. */
  --fp-accent: 220 90% 50%;
  --fp-accent-text: 220 100% 98%;
  --fp-ok: 142 70% 45%;
  --fp-warn: 38 92% 50%;
  --fp-err: 0 84% 60%;

  /* Dedicated badge-text tokens: same hue/saturation as the base tone, lightness
     lowered so full-strength text on the tone's own `/10` tinted background
     (e.g. `bg-fp-ok/10 text-fp-ok-text`) clears WCAG AA 4.5:1. The base
     --fp-ok/--fp-warn/--fp-err/--fp-accent tokens stay untouched — they still
     drive solid fills (StatusDot, chart series, buttons). Worst case measured
     against --fp-bg-2 (badges also render on row-hover/active surfaces, which
     tint slightly grayer than --fp-bg-1 and lower the ratio further):
     ok 142 70% 27%   → 4.86:1 on bg-2 tint (5.30:1 on bg-1 tint)
     warn 38 92% 28%  → 5.11:1 on bg-2 tint (5.57:1 on bg-1 tint)
     err 0 84% 42%    → 4.87:1 on bg-2 tint (5.32:1 on bg-1 tint)
     accent 220 90% 46% → 5.04:1 on bg-2 tint (5.50:1 on bg-1 tint) */
  --fp-ok-text: 142 70% 27%;
  --fp-warn-text: 38 92% 28%;
  --fp-err-text: 0 84% 42%;
  --fp-accent-badge-text: 220 90% 46%;

  /* Radius */
  --fp-radius-sm: 0.25rem;
  --fp-radius: 0.5rem;
  --fp-radius-lg: 0.75rem;

  /* Spacing base (4pt grid) */
  --fp-space-unit: 4px;

  /* Type */
  --fp-font-sans:
    ui-sans-serif, system-ui, -apple-system, Segoe UI, Roboto, Helvetica, Arial, sans-serif;
  --fp-font-mono: ui-monospace, SFMono-Regular, Menlo, Consolas, monospace;

  /* Motion */
  --fp-duration-fast: 120ms;
  --fp-duration: 180ms;
  --fp-duration-slow: 280ms;
  --fp-ease-out: cubic-bezier(0.22, 1, 0.36, 1);
  --fp-ease-in-out: cubic-bezier(0.65, 0, 0.35, 1);
  --fp-ease-spring: cubic-bezier(0.34, 1.56, 0.64, 1);
}

.dark {
  --fp-bg-1: 220 15% 10%;
  --fp-bg-2: 220 15% 14%;
  --fp-bg-3: 220 15% 18%;
  --fp-text-1: 220 10% 96%;
  --fp-text-2: 220 10% 78%;
  --fp-text-3: 220 10% 60%;
  --fp-border-1: 220 13% 22%;
  --fp-border-2: 220 13% 28%;

  /* Badge-text tokens re-checked against the dark surfaces above (worst case
     is --fp-bg-2, same as light). --fp-ok/--fp-warn at their original
     lightness already clear 4.5:1 in dark mode, so ok/warn carry the base
     tone through unchanged here — only err and accent needed re-lightening:
     ok 142 70% 45% (base, unchanged) → 5.82:1 on bg-2 tint (6.54:1 on bg-1 tint)
     warn 38 92% 50% (base, unchanged) → 6.26:1 on bg-2 tint (7.04:1 on bg-1 tint)
     err 0 84% 68%    → 4.84:1 on bg-2 tint (5.39:1 on bg-1 tint), base failed at 3.84:1
     accent 220 90% 68% → 5.07:1 on bg-2 tint (5.64:1 on bg-1 tint), base failed at 2.64:1 */
  --fp-ok-text: 142 70% 45%;
  --fp-warn-text: 38 92% 50%;
  --fp-err-text: 0 84% 68%;
  --fp-accent-badge-text: 220 90% 68%;
}

@media (prefers-reduced-motion: reduce) {

    --fp-duration: 0ms;
    --fp-duration-slow: 0ms;
  }
}
