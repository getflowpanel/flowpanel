/**
 * Ejected detail view for "{{name}}". Receives `[id]` from the route segment.
 */
import { getRow } from "../actions";

export default async function DetailPage({
  params,
}: {
  params: Promise<{ id: string }>;
}) {
  const { id } = await params;
  const row = await getRow(id);
  return (
    <div className="space-y-4 p-6">
      <h1 className="text-xl font-semibold">{{name}} · {id}</h1>
      <pre className="overflow-auto rounded-fp-lg border border-fp-border-1 bg-fp-bg-1 p-4 text-xs">
        {JSON.stringify(row, null, 2)}
      </pre>
    </div>
  );
}
