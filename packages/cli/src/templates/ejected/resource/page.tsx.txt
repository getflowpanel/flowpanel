// Ejected list view for "{{name}}": rows still pass the protected controller, so auth, field rules and tenant scope hold.
import { listRows } from "./actions";

export default async function Page() {
  const result = await listRows();
  return (
    <div className="space-y-4 p-6">
      <h1 className="text-xl font-semibold">{{name}}</h1>
      <pre className="overflow-auto rounded-fp-lg border border-fp-border-1 bg-fp-bg-1 p-4 text-xs">
        {JSON.stringify(result.rows, null, 2)}
      </pre>
    </div>
  );
}
