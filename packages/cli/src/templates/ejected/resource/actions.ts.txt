import "server-only";

import type { FlowpanelResult } from "@flowpanel/kit";
import { createFlowpanel } from "@flowpanel/kit/next";
import config from "{{CONFIG_IMPORT}}";

const flowpanel = createFlowpanel(config);

function unwrap<T>(result: FlowpanelResult<T>): T {
  if (!result.ok) throw new Error(result.error.message);
  return result.data;
}

async function resource() {
  const request = await flowpanel.request();
  return request.resource("{{name}}");
}

export async function listRows() {
  return unwrap(await (await resource()).list());
}

export async function getRow(id: string) {
  return unwrap(await (await resource()).get(id));
}

export async function createRow(input: Record<string, unknown>) {
  return unwrap(await (await resource()).create(input));
}

export async function updateRow(id: string, input: Record<string, unknown>) {
  return unwrap(await (await resource()).update(id, input));
}

export async function deleteRow(id: string) {
  return unwrap(await (await resource()).delete(id));
}
