/**
 * FlowPanel's session module, on the Lucia (v3+) preset.
 *
 * `flowpanel init` found lucia in package.json and wired this file to the
 * instance it detected. The preset reads Lucia's own session cookie and
 * validates it per request. flowpanel.config.ts imports `getSession` from here —
 * replace its `auth` block with `flowpanelAuth` to use the whole preset.
 */
import { withLucia } from "@flowpanel/kit/auth";
import { lucia } from "{{AUTH_MODULE}}";

export const flowpanelAuth = withLucia({
  lucia,
  requireRole: "admin",
});

export const getSession = flowpanelAuth.session;
