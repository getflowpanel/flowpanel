import { defineAdmin } from "@flowpanel/kit";
import { drizzleAdapter } from "@flowpanel/kit/drizzle";
import { db } from "{{DB}}";
import * as schema from "{{SCHEMA}}";
import { getSession } from "{{AUTH}}";

// Augment "@flowpanel/kit": it is the package you installed, and the
// interfaces below merge into the same declarations "@flowpanel/core" owns.
declare module "@flowpanel/kit" {
  interface FlowpanelTypes {
    db: typeof db;
    // Every table and view in the schema becomes a typed resource name and row.
    schema: typeof schema;
  }
  // Only for a resource renamed with options.name, e.g. one over a SQL view:
  // interface FlowpanelResources { user: typeof schema.adminUserRow.$inferSelect }
}

export default defineAdmin({
  paths: { admin: "{{ADMIN_PATH}}", api: "{{API_PATH}}" },
  adapter: drizzleAdapter({ db, schema }),

  auth: {
    session: getSession,
    role: (s) =>
      (s as { user?: { role?: string } } | null | undefined)?.user?.role ?? "guest",
    requireRole: "admin",
  },

  realtime: { driver: "memory" },

  theme: {
    brand: { name: "{{APP_NAME}}" },
    accent: "220 90% 50%",
    mode: "auto",
  },

  resources: [
    // Add `resource` to the "@flowpanel/kit" import above, then:

  ],
