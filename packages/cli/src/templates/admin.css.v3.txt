/* ---------------------------------------------------------------------------
   Motion. Radix primitives set data-state="open|closed" on their content and
   overlay nodes and keep the node mounted until the exit animation finishes,
   so a single fp-anim-* class gives a primitive both directions.
   Exits run faster than entries (~60%). */

@keyframes fp-fade-in {
  from {
    opacity: 0;
  }
}
@keyframes fp-fade-out {
  to {
    opacity: 0;
  }
}

@keyframes fp-pop-in {
  from {
    opacity: 0;
    transform: scale(0.97) translateY(-3px);
  }
}
@keyframes fp-pop-out {
  to {
    opacity: 0;
    transform: scale(0.98);
  }
}

/* Centred dialogs animate `scale`, never `transform`. Tailwind centres them
   with the standalone `translate` property, and a `transform: translate(…)`
   in a keyframe does not replace it — the two compose, so the panel spends the
   whole animation at -100%/-100% and snaps back when it ends. `translate`,
   `rotate` and `scale` are independent properties, so scaling here leaves the
   centring alone. Travel stays small: a wide jump reads as a snap. */
@keyframes fp-dialog-in {
  from {
    opacity: 0;
    scale: 0.97;
  }
  to {
    opacity: 1;
    scale: 1;
  }
}
@keyframes fp-dialog-out {
  from {
    opacity: 1;
    scale: 1;
  }
  to {
    opacity: 0;
    scale: 0.98;
  }
}

@keyframes fp-slide-in-right {
  from {
    transform: translateX(100%);
  }
}
@keyframes fp-slide-out-right {
  to {
    transform: translateX(100%);
  }
}
@keyframes fp-slide-in-left {
  from {
    transform: translateX(-100%);
  }
}
@keyframes fp-slide-out-left {
  to {
    transform: translateX(-100%);
  }
}
@keyframes fp-slide-in-bottom {
  from {
    transform: translateY(100%);
  }
}
@keyframes fp-slide-out-bottom {
  to {
    transform: translateY(100%);
  }
}
@keyframes fp-slide-in-top {
  from {
    transform: translateY(-100%);
  }
}
@keyframes fp-slide-out-top {
  to {
    transform: translateY(-100%);
  }
}

/* Dropdowns, popovers, selects, tooltips — scale from the trigger edge. */
.fp-anim-pop[data-state="open"],
.fp-anim-pop[data-state="delayed-open"] {
  animation: fp-pop-in var(--fp-duration) var(--fp-ease-out);
  transform-origin: var(
    --radix-popover-content-transform-origin,
    var(
      --radix-dropdown-menu-content-transform-origin,
      var(
        --radix-select-content-transform-origin,
        var(--radix-tooltip-content-transform-origin, center top)
      )
    )
  );
}
.fp-anim-pop[data-state="closed"] {
  animation: fp-pop-out var(--fp-duration-fast) var(--fp-ease-in-out) forwards;
}
/* Radix keeps content mounted until the exit animation reports `animationend`.
   Inside Popover's portal that animation never starts, so the panel stays in
   the DOM — invisible in places, but still painting and still swallowing the
   clicks underneath it. Opting Popover out of the exit lets Presence see no
   animation and unmount at once. Select/Dropdown/Dialog are unaffected. */
.fp-anim-pop.fp-anim-no-exit[data-state="closed"] {
  animation: none;
}

/* Centered dialogs — keyframes carry the centering translate because the
   animation's `transform` overrides the utility-composed one. */
/* `will-change` names exactly what the keyframes touch, so the panel gets its
   own layer and is composited instead of repainting against the scrim. */
.fp-anim-dialog[data-state="open"] {
  animation: fp-dialog-in var(--fp-duration-slow) var(--fp-ease-out);
  will-change: scale, opacity;
}
.fp-anim-dialog[data-state="closed"] {
  animation: fp-dialog-out var(--fp-duration) var(--fp-ease-in-out) forwards;
  will-change: scale, opacity;
}

/* Press feedback for buttons and tappable chips — transform only, so it
   never shifts layout. Longhands, so the same class also drives the color
   transition without fighting Tailwind's transition utilities. */
/* `box-shadow` is deliberately absent: Tailwind draws focus rings as box-shadow,
   and transitioning it makes the ring fade in — slow feedback for a keyboard
   affordance, and long enough to see the offset ring resolve to its colour. */
.fp-press {
  transition-property: color, background-color, border-color, opacity, transform;
  transition-duration: var(--fp-duration-fast);
  transition-timing-function: var(--fp-ease-out);
}
.fp-press:active {
  transform: scale(0.97);
}

/* Fades the clipped edges of a horizontal scroller so cut-off tabs read as
   "there is more" instead of as a rendering glitch. The side is driven by
   data-overflow, set from the scroll position, so a strip that fits — or one
   scrolled hard against an end — keeps that edge crisp. */
.fp-scroll-fade-x {
  --fp-scroll-fade: 1.5rem;
}
.fp-scroll-fade-x[data-overflow="start"] {
  mask-image: linear-gradient(to right, transparent 0, #000 var(--fp-scroll-fade));

  mask-image: linear-gradient(to left, transparent 0, #000 var(--fp-scroll-fade));
}

    to right,
    transparent 0,
    #000 var(--fp-scroll-fade),

  );
}
